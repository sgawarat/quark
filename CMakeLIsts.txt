cmake_minimum_required(VERSION 3.18)
project(quark)

set(QMK_FIRMWARE_DIR "" CACHE PATH "Path to the qmk_firmware directory")
set(QUARK_KEYBOARD "" CACHE STRING "Keyboard name")
set(QUARK_KEYMAP "" CACHE STRING "Keymap name")

if(NOT IS_DIRECTORY ${QMK_FIRMWARE_DIR})
    message(FATAL_ERROR "QMK Firmware Directory '${QMK_FIRMWARE_DIR}' NOT FOUND")
endif()

set(QUARK_CONFIG_HEADERS "")

# Keyboard
set(QUARK_KEYBOARD_DIR "${CMAKE_SOURCE_DIR}/keyboards/${QUARK_KEYBOARD}")

if(NOT EXISTS "${QUARK_KEYBOARD_DIR}/keymap.h")
    message(FATAL_ERROR "Keyboard '${QUARK_KEYBOARD}' NOT FOUND")
endif()

if(EXISTS "${QUARK_KEYBOARD_DIR}/config.h")
    list(APPEND QUARK_CONFIG_HEADERS ${QUARK_KEYBOARD_DIR}/config.h)
endif()

if(EXISTS "${QUARK_KEYBOARD_DIR}/CMakeLists.txt")
    add_subdirectory(${QUARK_KEYBOARD_DIR})
endif()

if(NOT TARGET keyboard)
    add_library(keyboard INTERFACE)
endif()

# Keymap
set(QUARK_KEYMAP_DIR "${QUARK_KEYBOARD_DIR}/keymaps/${QUARK_KEYMAP}")

if(NOT EXISTS "${QUARK_KEYMAP_DIR}/keymap.c")
    message(FATAL_ERROR "Keymap '${QUARK_KEYBOARD}' of the Keyboard '${QUARK_KEYBOARD}' NOT FOUND")
endif()

if(EXISTS "${QUARK_KEYMAP_DIR}/config.h")
    list(APPEND QUARK_CONFIG_HEADERS ${QUARK_KEYMAP_DIR}/config.h)
endif()

if(EXISTS "${QUARK_KEYMAP_DIR}/CMakeLists.txt")
    add_subdirectory(${QUARK_KEYMAP_DIR})
endif()

if(NOT TARGET keymap)
    add_library(keymap INTERFACE)
endif()

# config
add_library(config INTERFACE)
target_precompile_headers(config INTERFACE
    ${QUARK_CONFIG_HEADERS}
    ./include/quark/stable.h
)
target_compile_options(config INTERFACE
    -Wall
    -Wextra

    -Wno-c++98-compat
    -Wno-c++98-compat-pedantic
    -Wno-pre-c++17-compat
    -Wno-pre-c++20-compat

    -Wno-reserved-identifier
    -Wno-unused-parameter
    -Wno-gnu-include-next
    -Wno-gnu-anonymous-struct
    -Wno-gnu-zero-variadic-macro-arguments
    -Wno-gnu-folding-constant
    -Wno-pre-c11-compat
    -Wno-c23-extensions
    -Wno-c2y-extensions

    -Wno-missing-designated-field-initializers
    -Wno-covered-switch-default

    -Wno-padded
)
target_compile_features(config INTERFACE
    cxx_std_20
)
target_compile_definitions(config INTERFACE
    WIN32_LEAN_AND_MEAN
    NOMINMAX
    UNICODE
    _UNICODE
)
target_include_directories(config INTERFACE
    ${QMK_FIRMWARE_DIR}/quantum
    ${QMK_FIRMWARE_DIR}/quantum/keymap_extras
    ${QMK_FIRMWARE_DIR}/quantum/logging
    ${QMK_FIRMWARE_DIR}/quantum/nvm
    ${QMK_FIRMWARE_DIR}/quantum/nvm/eeprom
    ${QMK_FIRMWARE_DIR}/quantum/process_keycode
    ${QMK_FIRMWARE_DIR}/quantum/sequencer
    ${QMK_FIRMWARE_DIR}/platforms
    ${QMK_FIRMWARE_DIR}/tmk_core/protocol
    ./include
    ./include/quark/quantum
)

add_subdirectory(src)
add_subdirectory(platforms)
add_subdirectory(tools/key_test)
