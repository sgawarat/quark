cmake_minimum_required(VERSION 3.21)
project(tmk_desktop)

set(TMK_CORE_DIR "" CACHE PATH "Path to the tmk_core directory")
set(QMK_SOURCE_DIR "${TMK_CORE_DIR}")

if(NOT IS_ABSOLUTE ${TMK_CORE_DIR})
    set(TMK_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/${TMK_CORE_DIR}")
endif()

set(TMK_DESKTOP_LAYOUT_DIR "" CACHE PATH "Path to the layout directory")

if(NOT IS_ABSOLUTE ${TMK_DESKTOP_LAYOUT_DIR})
    set(TMK_DESKTOP_LAYOUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/${TMK_DESKTOP_LAYOUT_DIR}")
endif()

set(TMK_DESKTOP_KEYMAP_DIR "" CACHE PATH "Path to the keymap directory")

if(NOT IS_ABSOLUTE ${TMK_DESKTOP_KEYMAP_DIR})
    set(TMK_DESKTOP_KEYMAP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/${TMK_DESKTOP_KEYMAP_DIR}")
endif()

if(NOT EXISTS ${TMK_CORE_DIR})
    message(FATAL_ERROR "tmk_core directory '${TMK_CORE_DIR}' NOT FOUND")
endif()

if(NOT EXISTS ${TMK_DESKTOP_LAYOUT_DIR})
    message(FATAL_ERROR "layout directory '${TMK_DESKTOP_LAYOUT_DIR}' NOT FOUND")
endif()

if(NOT EXISTS ${TMK_DESKTOP_KEYMAP_DIR})
    message(FATAL_ERROR "keymap directory '${TMK_DESKTOP_KEYMAP_DIR}' NOT FOUND")
endif()

add_library(config INTERFACE)
target_precompile_headers(config INTERFACE
    ${TMK_DESKTOP_KEYMAP_DIR}/config.h
    ./include/tmk_desktop/stable.h
)
target_compile_options(config INTERFACE
    -Wall
    -Wextra

    # -Werror
    -Wno-c++98-compat
    -Wno-c++98-compat-pedantic
    -Wno-pre-c++17-compat
    -Wno-pre-c++20-compat

    -Wno-reserved-identifier
    -Wno-unused-parameter
    -Wno-gnu-include-next
    -Wno-gnu-anonymous-struct
    -Wno-gnu-zero-variadic-macro-arguments
    -Wno-gnu-folding-constant
    -Wno-pre-c11-compat
    -Wno-c23-extensions
    -Wno-c2y-extensions

    -Wno-missing-designated-field-initializers
    -Wno-covered-switch-default

    -Wno-padded
)
target_compile_features(config INTERFACE
    cxx_std_23
    c_std_23
)
target_compile_definitions(config INTERFACE
    KEYMAP_C="keymap.h"

    # Win32
    WIN32_LEAN_AND_MEAN
    NOMINMAX
    UNICODE
    _UNICODE
)
target_include_directories(config INTERFACE
    ${TMK_CORE_DIR}/quantum
    ${TMK_CORE_DIR}/quantum/keymap_extras
    ${TMK_CORE_DIR}/quantum/logging
    ${TMK_CORE_DIR}/quantum/nvm
    ${TMK_CORE_DIR}/quantum/nvm/eeprom
    ${TMK_CORE_DIR}/quantum/process_keycode
    ${TMK_CORE_DIR}/quantum/sequencer
    ${TMK_CORE_DIR}/platforms
    ${TMK_CORE_DIR}/tmk_core/protocol
    ./include
    ./include/tmk_desktop/quantum
)

add_subdirectory(src)
add_subdirectory(${TMK_DESKTOP_LAYOUT_DIR})
add_subdirectory(${TMK_DESKTOP_KEYMAP_DIR})
add_subdirectory(platforms)
add_subdirectory(tools/key_test)
